ASCIIDJ CONTROLS:
(note: code doesnt pull control schema from here)

modifier:
	hold LEFT SHIFT while touching any control to reset it to its default
	(defaults are the *_START_VAL entries in InputTuning.hpp)

mixer:
	left deck volume up: t
	left deck volume down: g
	right deck volume up: y
	right deck volume down: h
	crossfader to the left: b
	crossfader to the right: n

left equalizer:
	highs louder: r
	highs quieter: e
	mids louder: f
	mids quieter: d
	lows louder: v
	lows quieter: c

right equalizer:
	highs louder: i
	highs quieter: u
	mids louder: k
	mids quieter: j
	lows louder: ,
	lows quieter: m

left deck:
	speed up: w
	slow down: q
	spin forward: s
	spin back: a
	pause / play: x

right deck:
	speed up: p
	slow down: o
	spin forward: ;
	spin back: l
	pause / play: .

file browser:
	toggle file-select mode: `
	move selection up / down: arrow up / arrow down
	load selection onto left deck / right deck: arrow left / arrow right (only while in file-select mode)
